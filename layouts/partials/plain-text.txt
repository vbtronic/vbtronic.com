{{- /* Turn a page's rendered HTML into readable plain text: headings and list items
       keep Markdown-style markers, links keep their address, entities are decoded. */ -}}
{{- $s := .Content | string -}}
{{- $s = replaceRE `<a [^>]*href="(https?://[^"]+)"[^>]*>(.*?)</a>` "$2 ($1)" $s -}}
{{- $s = replaceRE `<a [^>]*href="(/[^"]*)"[^>]*>(.*?)</a>` (printf "$2 (%s$1)" (strings.TrimSuffix "/" site.BaseURL)) $s -}}
{{- $s = replaceRE `<h2[^>]*>` "\n\n## " $s -}}
{{- $s = replaceRE `<h[3-6][^>]*>` "\n\n### " $s -}}
{{- $s = replaceRE `</h[1-6]>` "\n" $s -}}
{{- $s = replaceRE `<li[^>]*>` "\n- " $s -}}
{{- $s = replaceRE `</p>|</ul>|</ol>|</blockquote>|</pre>` "\n\n" $s -}}
{{- $s = replaceRE `<br\s*/?>` "\n" $s -}}
{{- $s = replaceRE `<[^>]+>` "" $s -}}
{{- $s = htmlUnescape $s -}}
{{- $s = replaceRE `(?m)^[ \t]+|[ \t]+$` "" $s -}}
{{- $s = replaceRE `\n{3,}` "\n\n" $s -}}
{{- strings.TrimSpace $s -}}
